export function remainingDays(validTo, now = Date.now()) {

function toCert(obj) {
  if (!obj) return null;
  let str;
  if (obj instanceof X509Certificate) return obj;
  if (obj.rawCert) str = obj.rawCert.toString('utf8');
  else if (Buffer.isBuffer(obj)) str = obj.toString('utf8');
  else if (typeof obj === 'string') str = obj;
  else return null;
  if (!str.includes('BEGIN CERTIFICATE')) return null;
  try {
    return new X509Certificate(str);
  } catch {
    return null;
  }
}

export function fetchCertChain({ host, port = 443, timeoutMs = 5000, servername, rejectUnauthorized = true }) {
  const isIpLiteral = net.isIP(host) !== 0;
  const useSni = servername ?? (isIpLiteral ? undefined : host);
  return new Promise((resolve, reject) => {
    const sock = connect({
      host,
      port,
      ...(useSni ? { servername: useSni } : {}),
      rejectUnauthorized,
      minVersion: 'TLSv1',
    });
    const cleanup = () => { clearTimeout(timer); };
    const timer = setTimeout(() => {
      sock.destroy(new Error(`cert fetch timeout after ${timeoutMs}ms`));
    }, timeoutMs);
    sock.setTimeout(timeoutMs);
    sock.on('secureConnect', () => {
      setImmediate(() => {
        const chain = sock.getPeerCertificate(true);
        const certs = toCertList(chain);
        cleanup();
        sock.end();
        resolve({ certs });
      });
    });
    sock.on('timeout', () => {
      sock.destroy(new Error(`cert fetch timeout after ${timeoutMs}ms`));
    });
    sock.on('error', (e) => {
      cleanup();
      const err = new Error(`cert fetch failed: ${e.message}`);
      err.code = e.code || 'TLS';
      reject(err);
    });
  });
}

export function inspectPemChain(pem) {

function fmtAgo(ts) {
  if (!ts) return '—';

}

export function validateConfig(config) {
  const errors = [];

  const { http, tls } = config;
  if (http == null && tls == null) errors.push('config has no checks: add an `http` or `tls` array');
  if (http != null && !Array.isArray(http)) errors.push('config.http must be an array of http check entries');
  if (tls != null && !Array.isArray(tls)) errors.push('config.tls must be an array of tls check entries');
  if (Array.isArray(http)) {
    for (const [i, h] of http.entries()) {
      const where = `config.http[${i}]`;
      if (!h || typeof h !== 'object') { errors.push(`${where} must be an object`); continue; }
      if (!isNonEmptyString(h.url)) errors.push(`${where} is missing a URL (e.g. "https://example.com/health")`);
      if (h.assert != null && typeof h.assert !== 'object') errors.push(`${where}.assert must be an object`);
      if (h.assert?.bodyIncludes != null && !isNonEmptyString(h.assert.bodyIncludes)) {
